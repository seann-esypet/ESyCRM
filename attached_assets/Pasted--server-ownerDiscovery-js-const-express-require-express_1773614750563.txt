// ─── Name extraction via Claude ───────────────────────────────────────────────
async function extractNameWithClaude(textBlob, businessName) {
  if (!ANTHROPIC_API_KEY || !textBlob) return null;
  try {
    const res = await fetch('https://api.anthropic.com/v1/messages', {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'x-api-key': ANTHROPIC_API_KEY,
        'anthropic-version': '2023-06-01'
      },
      body: JSON.stringify({
        model: 'claude-sonnet-4-20250514',
        max_tokens: 100,
        messages: [{
          role: 'user',
          content: `You are extracting the first name of the business owner of "${businessName}" from web text snippets.

Text: """
${textBlob.slice(0, 2000)}
"""

Rules:
- Return ONLY the owner's first name as a single word, nothing else
- If no owner first name is clearly present, return the word: null
- Do not guess or infer — only return a name if it's explicitly stated as owner/founder/proprietor
- Ignore names of customers, reviewers, or employees who are not the owner`
        }]
      })
    });
    const data = await res.json();
    const raw = data?.content?.[0]?.text?.trim();
    if (!raw || raw.toLowerCase() === 'null') return null;
    // Validate it looks like a real first name (1-20 alpha chars)
    if (/^[A-Za-z]{1,20}$/.test(raw)) return raw;
    return null;
  } catch (e) {
    console.error('Claude extraction error:', e.message);
    return null;
  }
}

// ─── Method 1: SerpAPI Google Search ─────────────────────────────────────────
async function searchGoogle(businessName, city) {
  if (!SERP_API_KEY) return { name: null, source: null, confidence: null };

  const queries = [
    `"${businessName}" owner grooming`,
    `"${businessName}" "${city}" owner`,
    `"${businessName}" grooming "owned by"`,
    `"${businessName}" founder groomer`
  ];

  const ownerPatterns = [
    /owned\s+by\s+([A-Z][a-z]+)/i,
    /owner[:\s]+([A-Z][a-z]+)/i,
    /hi[,\s]+i'?m\s+([A-Z][a-z]+)/i,
    /meet\s+([A-Z][a-z]+)[,\s].{0,30}owner/i,
    /([A-Z][a-z]+)[,\s].{0,20}owner\s+of/i,
    /founder[:\s]+([A-Z][a-z]+)/i,
    /proprietor[:\s]+([A-Z][a-z]+)/i,
  ];

  for (const query of queries) {
    try {
      const url = `https://serpapi.com/search.json?q=${encodeURIComponent(query)}&api_key=${SERP_API_KEY}&num=5`;
      const res = await fetch(url);
      const data = await res.json();

      // Collect all text from snippets, knowledge graph, answer box
      const textChunks = [];
      if (data.answer_box?.answer) textChunks.push(data.answer_box.answer);
      if (data.answer_box?.snippet) textChunks.push(data.answer_box.snippet);
      if (data.knowledge_graph?.description) textChunks.push(data.knowledge_graph.description);
      (data.organic_results || []).slice(0, 4).forEach(r => {
        if (r.snippet) textChunks.push(r.snippet);
      });

      const combined = textChunks.join(' ');

      // Try regex patterns first (high confidence)
      for (const pattern of ownerPatterns) {
        const match = combined.match(pattern);
        if (match?.[1]) {
          return { name: match[1], source: 'Google Search', confidence: 'high' };
        }
      }

      // Fall back to Claude extraction (medium confidence)
      if (combined.length > 50) {
        const claudeName = await extractNameWithClaude(combined, businessName);
        if (claudeName) {
          return { name: claudeName, source: 'Google Search (AI)', confidence: 'verify' };
        }
      }
    } catch (e) {
      console.error('SerpAPI error:', e.message);
    }
  }

  return { name: null, source: null, confidence: null };
}

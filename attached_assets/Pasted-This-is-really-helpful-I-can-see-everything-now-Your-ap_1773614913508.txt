// server/ownerDiscovery.js
const express = require('express');
const router = express.Router();

const SERP_API_KEY = process.env.SERP_API_KEY;
const ANTHROPIC_API_KEY = process.env.ANTHROPIC_API_KEY;

// ─── Name extraction via Claude ───────────────────────────────────────────────
async function extractNameWithClaude(textBlob, businessName) {
  if (!ANTHROPIC_API_KEY || !textBlob) return null;
  try {
    const res = await fetch('https://api.anthropic.com/v1/messages', {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'x-api-key': ANTHROPIC_API_KEY,
        'anthropic-version': '2023-06-01'
      },
      body: JSON.stringify({
        model: 'claude-sonnet-4-20250514',
        max_tokens: 100,
        messages: [{
          role: 'user',
          content: `You are extracting the first name of the business owner of "${businessName}" from web text snippets.

Text: """
${textBlob.slice(0, 2000)}
"""

Rules:
- Return ONLY the owner's first name as a single word, nothing else
- If no owner first name is clearly present, return the word: null
- Do not guess or infer — only return a name if it's explicitly stated as owner/founder/proprietor
- Ignore names of customers, reviewers, or employees who are not the owner`
        }]
      })
    });
    const data = await res.json();
    const raw = data?.content?.[0]?.text?.trim();
    if (!raw || raw.toLowerCase() === 'null') return null;
    // Validate it looks like a real first name (1-20 alpha chars)
    if (/^[A-Za-z]{1,20}$/.test(raw)) return raw;
    return null;
  } catch (e) {
    console.error('Claude extraction error:', e.message);
    return null;
  }
}

// ─── Method 1: SerpAPI Google Search ─────────────────────────────────────────
async function searchGoogle(businessName, city) {
  if (!SERP_API_KEY) return { name: null, source: null, confidence: null };

  const queries = [
    `"${businessName}" owner grooming`,
    `"${businessName}" "${city}" owner`,
    `"${businessName}" grooming "owned by"`,
    `"${businessName}" founder groomer`
  ];

  const ownerPatterns = [
    /owned\s+by\s+([A-Z][a-z]+)/i,
    /owner[:\s]+([A-Z][a-z]+)/i,
    /hi[,\s]+i'?m\s+([A-Z][a-z]+)/i,
    /meet\s+([A-Z][a-z]+)[,\s].{0,30}owner/i,
    /([A-Z][a-z]+)[,\s].{0,20}owner\s+of/i,
    /founder[:\s]+([A-Z][a-z]+)/i,
    /proprietor[:\s]+([A-Z][a-z]+)/i,
  ];

  for (const query of queries) {
    try {
      const url = `https://serpapi.com/search.json?q=${encodeURIComponent(query)}&api_key=${SERP_API_KEY}&num=5`;
      const res = await fetch(url);
      const data = await res.json();

      // Collect all text from snippets, knowledge graph, answer box
      const textChunks = [];
      if (data.answer_box?.answer) textChunks.push(data.answer_box.answer);
      if (data.answer_box?.snippet) textChunks.push(data.answer_box.snippet);
      if (data.knowledge_graph?.description) textChunks.push(data.knowledge_graph.description);
      (data.organic_results || []).slice(0, 4).forEach(r => {
        if (r.snippet) textChunks.push(r.snippet);
      });

      const combined = textChunks.join(' ');

      // Try regex patterns first (high confidence)
      for (const pattern of ownerPatterns) {
        const match = combined.match(pattern);
        if (match?.[1]) {
          return { name: match[1], source: 'Google Search', confidence: 'high' };
        }
      }

      // Fall back to Claude extraction (medium confidence)
      if (combined.length > 50) {
        const claudeName = await extractNameWithClaude(combined, businessName);
        if (claudeName) {
          return { name: claudeName, source: 'Google Search (AI)', confidence: 'verify' };
        }
      }
    } catch (e) {
      console.error('SerpAPI error:', e.message);
    }
  }

  return { name: null, source: null, confidence: null };
}

// ─── Method 2: Website Scraper ────────────────────────────────────────────────
async function scrapeWebsite(websiteUrl, businessName) {
  if (!websiteUrl) return { name: null, source: null, confidence: null };

  // Normalize URL
  let base = websiteUrl.trim();
  if (!base.startsWith('http')) base = 'https://' + base;
  base = base.replace(/\/$/, '');

  const pagesToTry = ['', '/about', '/about-us', '/our-team', '/meet-us', '/contact', '/our-story'];

  const ownerPatterns = [
    /owned\s+by\s+([A-Z][a-z]+)/i,
    /hi[,\s]+i'?m\s+([A-Z][a-z]+)/i,
    /meet\s+([A-Z][a-z]+)[,\s].{0,30}owner/i,
    /([A-Z][a-z]+)[,\s]+owner\s+(?:and|&|of)/i,
    /founder[:\s]+([A-Z][a-z]+)/i,
    /<title>.*?([A-Z][a-z]+).*?(?:owner|grooming)<\/title>/i,
  ];

  for (const path of pagesToTry) {
    try {
      const url = base + path;
      const res = await fetch(url, {
        signal: AbortSignal.timeout(8000),
        headers: { 'User-Agent': 'Mozilla/5.0 (compatible; EsyCRM/1.0)' }
      });
      if (!res.ok) continue;

      const html = await res.text();

      // Strip HTML tags for cleaner text
      const text = html
        .replace(/<script[^>]*>[\s\S]*?<\/script>/gi, '')
        .replace(/<style[^>]*>[\s\S]*?<\/style>/gi, '')
        .replace(/<[^>]+>/g, ' ')
        .replace(/\s+/g, ' ');

      // Check og:description and meta description
      const ogDesc = html.match(/og:description[^>]*content="([^"]+)"/i)?.[1] || '';
      const metaDesc = html.match(/meta[^>]*name="description"[^>]*content="([^"]+)"/i)?.[1] || '';

      const allText = [ogDesc, metaDesc, text].join(' ');

      // Try regex patterns
      for (const pattern of ownerPatterns) {
        const match = allText.match(pattern);
        if (match?.[1]) {
          return { name: match[1], source: `Website (${path || 'homepage'})`, confidence: 'high' };
        }
      }

      // Claude fallback on about pages (more likely to have owner info)
      if (path && allText.length > 100) {
        const claudeName = await extractNameWithClaude(allText.slice(0, 3000), businessName);
        if (claudeName) {
          return { name: claudeName, source: `Website (${path})`, confidence: 'verify' };
        }
      }
    } catch (e) {
      // Timeout or fetch error — skip this page
    }
  }

  return { name: null, source: null, confidence: null };
}

// ─── Method 3: Email prefix heuristic ────────────────────────────────────────
function checkEmailPrefix(email, businessName) {
  if (!email) return { name: null, source: null, confidence: null };
  
  const genericPrefixes = ['info', 'hello', 'contact', 'admin', 'support', 'hi', 'mail',
    'office', 'team', 'grooming', 'salon', 'help', 'noreply'];
  
  const prefix = email.split('@')[0].toLowerCase().replace(/[^a-z]/g, '');
  
  if (genericPrefixes.some(g => prefix.startsWith(g))) return { name: null, source: null, confidence: null };
  
  // If prefix looks like a first name (4-12 letters, not matching biz name keywords)
  const bizWords = businessName.toLowerCase().split(/\s+/).map(w => w.replace(/[^a-z]/g, ''));
  if (prefix.length >= 3 && prefix.length <= 12 && !bizWords.some(w => prefix.includes(w.slice(0, 4)))) {
    const capitalized = prefix.charAt(0).toUpperCase() + prefix.slice(1);
    return { name: capitalized, source: 'Email prefix', confidence: 'low' };
  }
  
  return { name: null, source: null, confidence: null };
}

// ─── Main Discovery Route ─────────────────────────────────────────────────────
router.post('/api/contacts/:id/discover-owner', async (req, res) => {
  const { businessName, website, email, city } = req.body;

  if (!businessName) {
    return res.status(400).json({ error: 'businessName is required' });
  }

  const results = [];

  // Run methods in parallel where possible, then evaluate
  try {
    const [googleResult, websiteResult] = await Promise.all([
      searchGoogle(businessName, city || ''),
      scrapeWebsite(website, businessName)
    ]);

    if (googleResult.name) results.push(googleResult);
    if (websiteResult.name) results.push(websiteResult);

    // Email heuristic (instant, no API needed)
    const emailResult = checkEmailPrefix(email, businessName);
    if (emailResult.name) results.push(emailResult);

    // Return best result (priority: high > verify > low)
    const priority = { high: 0, verify: 1, low: 2 };
    results.sort((a, b) => priority[a.confidence] - priority[b.confidence]);

    if (results.length > 0) {
      return res.json({ found: true, ...results[0], allResults: results });
    }

    return res.json({ found: false, name: null, source: null, confidence: null });
  } catch (e) {
    console.error('Discovery error:', e);
    res.status(500).json({ error: 'Discovery failed', details: e.message });
  }
});

// client/components/OwnerNameFinder.jsx
import { useState } from "react";

const CONFIDENCE_CONFIG = {
  high: {
    label: "High Confidence",
    color: "bg-green-50 border-green-200",
    badge: "bg-green-100 text-green-800",
    icon: "✓"
  },
  verify: {
    label: "Verify Before Sending",
    color: "bg-yellow-50 border-yellow-200",
    badge: "bg-yellow-100 text-yellow-800",
    icon: "⚠"
  },
  low: {
    label: "Low Confidence",
    color: "bg-orange-50 border-orange-200",
    badge: "bg-orange-100 text-orange-800",
    icon: "?"
  }
};

export function OwnerNameFinder({ contact, onNameAccepted }) {
  const [status, setStatus] = useState("idle"); // idle | searching | found | not_found | error
  const [result, setResult] = useState(null);

  const handleSearch = async () => {
    setStatus("searching");
    setResult(null);

    try {
      const res = await fetch(`/api/contacts/${contact.id}/discover-owner`, {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({
          businessName: contact.company || contact.name,
          website: contact.website || "",
          email: contact.email || "",
          city: contact.city || ""
        })
      });

      if (!res.ok) throw new Error("Server error");
      const data = await res.json();

      if (data.found) {
        setResult(data);
        setStatus("found");
      } else {
        setStatus("not_found");
      }
    } catch (e) {
      console.error(e);
      setStatus("error");
    }
  };

  const handleAccept = () => {
    if (result?.name) {
      onNameAccepted(result.name);
      setStatus("idle");
      setResult(null);
    }
  };

  const handleDismiss = () => {
    setStatus("idle");
    setResult(null);
  };

  return (
    <div className="mt-2 mb-4">
      {/* Search Button */}
      {(status === "idle" || status === "error") && (
        <button
          type="button"
          onClick={handleSearch}
          className="inline-flex items-center gap-2 text-sm text-blue-600 border border-dashed border-blue-300 rounded-md px-3 py-1.5 hover:bg-blue-50 hover:border-blue-400 transition-colors"
        >
          <span>🔍</span>
          Search for Owner's First Name
        </button>
      )}

      {status === "error" && (
        <p className="text-xs text-red-500 mt-1">Search failed. Check API keys and try again.</p>
      )}

      {/* Loading State */}
      {status === "searching" && (
        <div className="flex items-center gap-2 text-sm text-muted-foreground py-2">
          <svg className="animate-spin h-4 w-4 text-blue-500" xmlns="http://www.w3.org/2000/svg" fill="none" viewBox="0 0 24 24">
            <circle className="opacity-25" cx="12" cy="12" r="10" stroke="currentColor" strokeWidth="4" />
            <path className="opacity-75" fill="currentColor" d="M4 12a8 8 0 018-8v8z" />
          </svg>
          <span>Searching Google, website, and email...</span>
        </div>
      )}

      {/* Found Result */}
      {status === "found" && result && (() => {
        const cfg = CONFIDENCE_CONFIG[result.confidence] || CONFIDENCE_CONFIG.verify;
        return (
          <div className={`rounded-lg border p-3 mt-2 ${cfg.color}`}>
            <div className="flex items-start justify-between gap-3">
              <div>
                <div className="flex items-center gap-2 mb-1">
                  <span className="font-semibold text-foreground text-base">{result.name}</span>
                  <span className={`text-xs px-2 py-0.5 rounded-full font-medium ${cfg.badge}`}>
                    {cfg.icon} {cfg.label}
                  </span>
                </div>
                <p className="text-xs text-muted-foreground">
                  Found via: <span className="font-medium">{result.source}</span>
                </p>
                {result.confidence !== "high" && (
                  <p className="text-xs text-yellow-700 mt-1">
                    Please verify this name before sending emails.
                  </p>
                )}
              </div>
            </div>

            <div className="flex gap-2 mt-3">
              <button
                type="button"
                onClick={handleAccept}
                className="inline-flex items-center gap-1 text-xs bg-green-600 hover:bg-green-700 text-white rounded px-3 py-1.5 font-medium transition-colors"
              >
                ✓ Use "{result.name}"
              </button>
              <button
                type="button"
                onClick={handleDismiss}
                className="inline-flex items-center text-xs border border-gray-300 hover:bg-gray-50 text-gray-600 rounded px-3 py-1.5 transition-colors"
              >
                Dismiss
              </button>
              <button
                type="button"
                onClick={handleSearch}
                className="inline-flex items-center text-xs text-blue-500 hover:underline px-1 py-1.5"
              >
                Search again
              </button>
            </div>
          </div>
        );
      })()}

      {/* Not Found */}
      {status === "not_found" && (
        <div className="rounded-lg border border-amber-200 bg-amber-50 p-3 mt-2">
          <p className="text-sm text-amber-800 font-medium">⚠ Owner name not found automatically</p>
          <p className="text-xs text-amber-700 mt-0.5">
            Try Googling: <span className="font-mono">"{contact.company || contact.name}" owner name</span>
          </p>
          <button
            type="button"
            onClick={handleDismiss}
            className="text-xs text-gray-500 hover:underline mt-2"
          >
            Dismiss
          </button>
        </div>
      )}
    </div>
  );
}

Key things this does that the Replit agent's version was missing:
The backend runs three methods simultaneously using Promise.all so it's fast.
